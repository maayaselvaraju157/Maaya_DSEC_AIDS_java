2)Insertion
import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
            Scanner sc = new Scanner(System.in);
        int a[]=new int[5];
        a[0]=10;
        a[1]=20;
        a[2]=30;
        a[3]=40;
        int n=4;
        System.out.println("Enter Value:");
        int value=sc.nextInt();
        System.out.println("Enter index (0 to "+n+"):");
        int pos = sc.nextInt();
        if(pos<0||pos>n||n==a.length){
            System.out.println("Invalid insertion");
        }else{
            for(int i=n;i>pos;i--){
                a[i]=a[i-1];
            }
            a[pos]=value;
            n++;
            for(int i=0; i<n;i++){
                System.out.println(a[i]+"");
            }
        }
        sc.close();
    }
}

6)Search(Binary search)

import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        int a[] = {40, 10, 30, 20, 50, 60, 35};
        int key = 60;

        Arrays.sort(a);

        int low = 0;
        int high = a.length - 1;
        int pos = -1;

        while (low <= high) {
            int mid = low + (high - low) / 2;

            if (a[mid] == key) {
                pos = mid;
                break;
            } else if (a[mid] < key) {
                low = mid + 1;
            } else {
                high = mid - 1;
            }
        }

        if (pos != -1) {
            System.out.println("Found at index: " + pos);
        } else {
            System.out.println("Not Found");
        }
    }
}

SHIFT
1)Shift first number to last
public class Main {
    public static void main(String[] args) {
        int a[] = {10, 20, 30, 40, 50};

        int shift = a[0]; 

        for (int i = 0; i < a.length - 1; i++) {
            a[i] = a[i + 1];
        }

        a[a.length - 1] = shift;

        for (int i = 0; i < a.length; i++) {
            System.out.println(a[i]);
        }
    }
}


2)Shift last number to first
public class Main {
    public static void main(String[] args) {
        int a[] = {10, 20, 30, 40, 50};

        int shift = a[a.length - 1];

        for (int i = a.length - 1; i > 0; i--) {
            a[i] = a[i - 1];
        }

        a[0] = shift;

        for (int i = 0; i < a.length; i++) {
            System.out.println(a[i]);
        }
    }
}
